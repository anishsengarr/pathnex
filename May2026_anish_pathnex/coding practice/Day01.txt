## Ansible Task — Install Nginx on Pathnex Server


- name: Install Nginx on Pathnex server
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present

## Kubernetes Task — Create Nginx Pod

apiVersion : v1
kind: pod
metadata:
   name: nex-pod
spec:
   containers:
    -name: web
    image: nginx:latest
    ports:
    -containerPort:80

## Jenkins Pipeline — Checkout Git

pipeline {
    agent any
    stages {
        stage('Checkout') {
            steps {
                git branch: 'main', url: 'https://github.com/Pathnex/app.git'
            }
        }
        stage('List Files') {
            steps {
                sh 'ls -la'
            }
        }
    }
}


## GitLab CI — Checkout Git


stages:
